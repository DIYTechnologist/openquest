cmake_minimum_required(VERSION 3.16)
project(openquest_controller_tracking LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc calib3d)
add_library(oq_controller_tracker STATIC src/tracker.cpp)
target_include_directories(oq_controller_tracker PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(oq_controller_tracker PUBLIC ${OpenCV_LIBS})
target_compile_options(oq_controller_tracker PRIVATE -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
foreach(program replay service selftest live)
    add_executable(oq-controller-${program} src/${program}.cpp)
    target_link_libraries(oq-controller-${program} PRIVATE oq_controller_tracker Threads::Threads)
endforeach()
enable_testing()
add_test(NAME controller_core COMMAND oq-controller-selftest)

add_executable(oq-controller-socket-test tests/socket_test.cpp)
target_include_directories(oq-controller-socket-test PRIVATE src)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME controller_socket COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_socket_test.py $<TARGET_FILE:oq-controller-service> $<TARGET_FILE:oq-controller-socket-test>)

add_executable(oq-controller-input-test tests/live_input_test.cpp)
target_link_libraries(oq-controller-input-test PRIVATE oq_controller_tracker)
add_test(NAME controller_live_input COMMAND oq-controller-input-test)
